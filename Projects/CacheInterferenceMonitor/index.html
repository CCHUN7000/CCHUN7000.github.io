Core Metrics

Preemption Related

  • Preempt count / sec
  • Preempt frequency by priority
  • → tracepoint: sched/sched_switch

Cache Related

  • L1-dcache-load-misses / loads
  • LLC-load-misses / loads
  • → Hardware PMU (perf_event)

CPI Related

  • CPI = Cycles / Instructions
  • Branch mispredictions
  • → Hardware PMU

Migration Related

  • Cross-core migration count
  • Cache state after migration
  • → tracepoint: sched/sched_migrate

    __uint(max_entries, 256);
    __type(key, u32);  // cpu_id

// Task cache statistics
struct task_cache_stats {
    u64 l1_misses, l1_hits, llc_misses, llc_hits;
    u64 cycles, instructions;
    u64 preempt_count, preempt_by_higher;
    u64 cross_core_migrate, runtime_ns;
};

sched_switch Handler

SEC("tp_btf/sched_switch")
int handle_sched_switch(struct pt_regs *ctx) {
    u64 ts = bpf_ktime_get_ns();
    u32 prev_pid = 0, next_pid = 0;  // from PT_REGS_PARM*

    // Get prev task runtime
    struct task_runtime_record *rec = bpf_map_lookup_elem(&task_runtime_record, &prev_pid);
    if (rec) {
        u64 runtime = ts - rec->last_start_ns;

        // Update histogram by runtime bucket
        u32 bucket = runtime < 1_000_000 ? 0 : runtime < 5_000_000 ? 1 : 2;
        u64 *count = bpf_map_lookup_elem(&preempt_histogram, &bucket);
        if (count) __sync_fetch_and_add(count, 1);

        // Short runtime = high priority preemption
        if (runtime < 1_000_000) {
            struct task_cache_stats *s = bpf_map_lookup_elem(&task_cache_stats, &prev_pid);
            if (s) __sync_fetch_and_add(&s->preempt_by_higher, 1);
        }
    }
    return 0;
}

// Python: Calculate interference score
def interference_score(self) -> float:
    preempt_score = min(30, self.preempt_by_higher * 2)
    migrate_score = min(30, self.cross_core_migrate * 5)
    cache_score = min(40, self.l1_miss_rate * 100)
    return preempt_score + migrate_score + cache_score

# Score Range:
#   0-25:  LOW       (acceptable)
#   25-50: MODERATE  (monitor)
#   50-100: HIGH      (action needed)

Quick Start

# 1. Compile eBPF program
clang -target bpf -O2 -Wall -c cache_interference.bpf.c -o cache_interference.bpf.o

# 2. Load and attach
bpftool prog load cache_interference.bpf.o /sys/fs/bpf/cache_interference
bpftool tracepoint attach sched sched_switch /sys/fs/bpf/cache_interference

# 3. Run monitor
python3 cache_interference_monitor.py

# 4. View results
bpftool map dump id 1  # task_cache_stats
