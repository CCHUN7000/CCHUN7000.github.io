Core Data Structures

IO Metrics (10 metrics, fixed-point ×100)

struct io_metrics {
    u64 iowait_percent;      /* iowait% ×100 */
    u64 read_latency_ns;    /* avg read latency in ns */
    u64 write_latency_ns;   /* avg write latency in ns */
    u64 read_queue_depth;   /* avg read queue depth ×100 */
    u64 write_queue_depth;  /* avg write queue depth ×100 */
    u64 utilization_percent; /* disk utilization% ×100 */
    u64 read_throughput;    /* read bytes/sec */
    u64 write_throughput;   /* write bytes/sec */
    u64 merge_rate;         /* merge rate ×100 */
    u64 avg_io_size;        /* average IO size in sectors ×100 */
};

Snapshot & Delta Pattern

struct io_snapshot {
    u64 timestamp_ns;
    u64 cpu_iowait, cpu_total;
    u64 nsecs[2], sectors[2], ios[2], merges[2];  /* read/write */
    u64 io_ticks;
    local_t in_flight[2];
};

// Delta = curr - prev (for rate calculations)
static void calc_delta(const struct io_snapshot *prev,
                       const struct io_snapshot *curr,
                       struct io_delta *delta)
{
    delta->time_delta_ns = curr->timestamp_ns - prev->timestamp_ns;
    delta->iowait_delta = curr->cpu_iowait - prev->cpu_iowait;
    // ... per-CPU aggregation via for_each_possible_cpu()
}

Snapshot Collection

static int take_snapshot(struct io_snapshot *snap)
{
    struct disk_stats *stat;
    unsigned int cpu;
    int i;

    memset(snap, 0, sizeof(*snap));
    snap->timestamp_ns = ktime_get_raw_ns();

    /* Get CPU iowait stats */
    kcpustat_cpu_fetch(&snap->cpu_iowait, target_cpu);

    /* Aggregate per-CPU block device stats */
    part_stat_lock();
    for_each_possible_cpu(cpu) {
        stat = per_cpu_ptr(monitor_bdev->bd_stats, cpu);
        for (i = 0; i < 2; i++) {  /* 0=read, 1=write */
            snap->nsecs[i] += stat->nsecs[i];
            snap->sectors[i] += stat->sectors[i];
            snap->ios[i] += stat->ios[i];
            snap->merges[i] += stat->merges[i];
        }
        snap->io_ticks += stat->io_ticks;
    }
    part_stat_unlock();
    return 0;
}

Metrics Calculation

static void calc_metrics(const struct io_delta *delta, struct io_metrics *out)
{
    /* iowait% = iowait_delta / total_delta × 10000 */
    if (delta->total_delta > 0)
        out->iowait_percent = (delta->iowait_delta * 10000) / delta->total_delta;

    /* Read latency = nsecs / ios */
    if (delta->ios_delta[0] > 0) {
        out->read_latency_ns = delta->nsecs_delta[0] / delta->ios_delta[0];
        out->read_throughput = (delta->sectors_delta[0] << 9) * 1000000000 /
                               max(delta->time_delta_ns, 1ULL);
        out->read_queue_depth = (delta->nsecs_delta[0] * 100) /
                                 max(delta->time_delta_ns, 1ULL);
        out->utilization_percent = (delta->io_ticks_delta * 10000) /
                                  max(delta->time_delta_ns / 1000, 1ULL);
    }
    /* Write metrics follow same pattern (index 1) */

    /* Merge rate% = merges / (ios + merges) × 10000 */
    divisor = delta->ios_delta[0] + delta->merges_delta[0];
    if (divisor > 0)
        out->merge_rate = (delta->merges_delta[0] * 10000) / divisor;
}

Procfs Interface

Trigger Handler

static ssize_t trigger_write(struct file *file, const char __user *buffer,
                             size_t count, loff_t *pos)
{
    struct io_snapshot curr;
    struct io_delta delta;
    struct io_metrics metrics;

    take_snapshot(&curr);

    if (!baseline_valid) {
        baseline_snapshot = curr;
        baseline_valid = true;
        return count;
    }

    /* Second trigger: calculate delta from last */
    calc_delta(&last_snapshot, &curr, &delta);
    calc_metrics(&delta, &metrics);
    ring_push_buf(&entry);

    last_snapshot = curr;
    return count;
}

Procfs Registration

static const struct proc_ops trigger_proc_ops = {
    .proc_write = trigger_write,
};

static const struct proc_ops data_proc_ops = {
    .proc_open = data_open,
    .proc_read = seq_read,
    .proc_lseek = seq_lseek,
    .proc_release = single_release,
};

proc_dir = proc_mkdir("io_monitor", NULL);
trigger_file = proc_create("trigger", 0222, proc_dir, &trigger_proc_ops);
data_file = proc_create("data", 0444, proc_dir, &data_proc_ops);

Usage

# Load module (default: sda)
sudo insmod io_monitor.ko
sudo insmod io_monitor.ko target_device=sdb

# Trigger snapshots
echo 1 > /proc/io_monitor/trigger  # First: set baseline
echo 1 > /proc/io_monitor/trigger  # Second: show delta

# View data
cat /proc/io_monitor/data

# Output example:
# IO Monitor Data (samples: 1)
# Index  Elapsed  iowait%  read_lat   write_lat  rd_qd  wr_qd  util%   ios
