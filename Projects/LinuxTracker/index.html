Key Algorithms

Commit Classification

def classify_commit(commit_msg: str) -> Dict[str, Any]:
    """Classify commit type by keywords in message"""
    msg_lower = commit_msg.lower()
    commit_type = {
        'feature': any(k in msg_lower for k in ['feature', 'add', 'implement', 'new']),
        'bugfix': any(k in msg_lower for k in ['fix', 'bug', 'repair', 'correct']),
        'refactor': any(k in msg_lower for k in ['refactor', 'clean', 'simplify', 'improve']),
        'performance': any(k in msg_lower for k in ['perf', 'optimize', 'speed', 'efficient']),
        'api': any(k in msg_lower for k in ['api', 'syscall', 'interface']),
        'documentation': any(k in msg_lower for k in ['doc', 'comment', 'readme']),
    }
    return {'type': commit_type, 'raw': commit_msg}

Cross-Reference (Jaccard Similarity)

def jaccard_similarity(set1: Set[str], set2: Set[str]) -> float:
    """Calculate Jaccard similarity between two token sets"""
    if not set1 or not set2:
        return 0.0
    intersection = len(set1 & set2)
    union = len(set1 | set2)
    return intersection / union if union > 0 else 0.0

def cross_reference(commits, lkml_results, ...):
    """Link commits to LKML discussions using Jaccard similarity"""
    commit_subjects = {c.hash: tokenize(c.subject) for c in commits}

    for lkml in lkml_results:
        lkml_tokens = tokenize(lkml.subject)
        best_score = 0.0
        for commit_hash, commit_tokens in commit_subjects.items():
            score = jaccard_similarity(lkml_tokens, commit_tokens)
            if score > best_score and score >= 0.3:
                best_match = commit_hash
                best_score = score

Usage Examples

Quick Summary (Offline)

# Analyze scheduler commits between versions
python -m tools.patch_analyzer summary \
    --path /path/to/linux \
    --subsystem kernel/sched/ \
    --from v6.1 \
    --to v6.8 \
    --max 50

# Output: JSON with commit list and metadata
{
  "subsystem": "kernel/sched/",
  "from": "v6.1",
  "to": "v6.8",
  "commits": [...],
  "meta": {
    "keywords": ["sched", "entity", "pick", "next", ...],
    "commit_count": 1234
  }
}

Full Analysis (With Network)

# Full analysis with LKML, Patchwork, and papers
python -m tools.patch_analyzer analyze \

    --max 20
